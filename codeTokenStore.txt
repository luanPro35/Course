// Ví dụ về file apiClient.js ở phía Frontend

import axios from 'axios';

// Tạo một "người giao hàng" axios riêng
const apiClient = axios.create({
baseURL: 'http://localhost:8080/api/v1', // URL gốc của BE
});

// --- 1. Request Interceptor ---
// Tự động gắn accessToken vào mọi request gửi đi
apiClient.interceptors.request.use(
(config) => {
const token = localStorage.getItem('accessToken');
if (token) {
config.headers['Authorization'] = `Bearer ${token}`;
}
return config;
},
(error) => {
return Promise.reject(error);
}
);

// --- 2. Response Interceptor ---
// Xử lý khi token hết hạn (lỗi 401)
apiClient.interceptors.response.use(
(response) => {
// Nếu response thành công, cứ trả về bình thường
return response;
},
async (error) => {
const originalRequest = error.config;

    // Nếu lỗi là 401 và chưa thử lại lần nào
    if (error.response.status === 401 && !originalRequest._retry) {
      originalRequest._retry = true; // Đánh dấu là đã thử lại

      try {
        // Gọi API /refresh để lấy accessToken mới
        const refreshToken = localStorage.getItem('refreshToken');
        const rs = await apiClient.post('/auth/refresh', { refreshToken });

        const { accessToken } = rs.data;

        // Lưu accessToken mới và gọi lại request cũ
        localStorage.setItem('accessToken', accessToken);
        return apiClient(originalRequest);
      } catch (_error) {
        // Nếu refresh token cũng thất bại (hết hạn hoặc không hợp lệ)
        // -> Xóa token, đẩy người dùng về trang đăng nhập
        localStorage.removeItem('accessToken');
        localStorage.removeItem('refreshToken');
        window.location.href = '/auth/login';
        return Promise.reject(_error);
      }
    }

    return Promise.reject(error);
}
);

export default apiClient;
